### Part B — ThemeManager.gd

Replace stub `scripts/autoloads/ThemeManager.gd`:

```gdscript
extends Node

signal theme_changed(palette: Dictionary)

const PALETTES: Dictionary = {
    "default": {
        "bg":       Color(0.055, 0.039, 0.118),
        "text":     Color.WHITE,
        "pile_warn":Color(1.0, 0.2, 0.2),
        "cols":     [
            Color(0.0,   0.784, 0.722),
            Color(0.0,   0.878, 1.000),
            Color(1.0,   0.757, 0.027),
            Color(0.694, 0.404, 1.000),
            Color(0.0,   0.941, 0.502),
        ]
    },
    # Ocean, Ember, Void are stubs — same palette as default until IAP unlocks them
    "ocean": { "bg": Color(0.02, 0.05, 0.15), "text": Color.WHITE,
               "pile_warn": Color(1.0, 0.2, 0.2),
               "cols": [Color(0.0, 0.5, 1.0), Color(0.0, 0.8, 0.9),
                        Color(0.2, 0.9, 0.6), Color(0.3, 0.6, 1.0), Color(0.0, 0.7, 0.8)] },
    "ember": { "bg": Color(0.10, 0.03, 0.01), "text": Color(1.0, 0.95, 0.8),
               "pile_warn": Color(1.0, 0.1, 0.0),
               "cols": [Color(1.0, 0.3, 0.0), Color(1.0, 0.6, 0.0),
                        Color(1.0, 0.8, 0.1), Color(0.9, 0.2, 0.1), Color(1.0, 0.5, 0.1)] },
    "void":  { "bg": Color(0.0, 0.0, 0.0), "text": Color(0.8, 0.8, 1.0),
               "pile_warn": Color(0.8, 0.0, 1.0),
               "cols": [Color(0.5, 0.0, 1.0), Color(0.8, 0.0, 0.8),
                        Color(0.3, 0.0, 0.8), Color(0.6, 0.2, 1.0), Color(0.4, 0.0, 0.6)] },
}

var _current: String = "default"

func _ready() -> void:
    _current = SaveGame.get_theme()

func current_palette() -> Dictionary:
    return PALETTES.get(_current, PALETTES["default"])

func column_colour(col_id: int) -> Color:
    return current_palette()["cols"][clamp(col_id, 0, 4)]

func apply_theme(name: String) -> void:
    if not PALETTES.has(name):
        return
    _current = name
    SaveGame.set_theme(name)
    emit_signal("theme_changed", current_palette())
```

### Part C — SettingsOverlay.gd

Create `scripts/SettingsOverlay.gd`:

#### 1. `_ready()`
```gdscript
$DimBackground.gui_input.connect(_on_dim_clicked)
$CloseButton.pressed.connect(queue_free)
$Panel/VBox/SoundRow/SoundToggle.button_pressed = SaveGame.get_sound_enabled()
$Panel/VBox/SoundRow/SoundToggle.toggled.connect(func(on):
    AudioManager.set_enabled(on)
    SaveGame.set_sound_enabled(on)
)
_setup_theme_buttons()
```

#### 2. `_setup_theme_buttons()`
```gdscript
func _setup_theme_buttons() -> void:
    var current := SaveGame.get_theme()
    for btn_name in ["ThemeDefault", "ThemeOcean", "ThemeEmber", "ThemeVoid"]:
        var btn: Button = $Panel/VBox/ThemeGrid.get_node(btn_name)
        var theme_key := btn_name.to_lower().replace("theme", "")
        btn.pressed.connect(func(): ThemeManager.apply_theme(theme_key))
        # Highlight current theme
        btn.flat = (theme_key != current)
        # Lock non-default themes (IAP gates implemented in T025)
        if theme_key != "default":
            btn.disabled = not SaveGame.is_theme_unlocked(theme_key)
```

#### 3. `_on_dim_clicked(event)`
```gdscript
func _on_dim_clicked(event: InputEvent) -> void:
    if event is InputEventScreenTouch and event.pressed:
        queue_free()
```

- Tap Settings button on StartScreen: SettingsOverlay slides in (or appears)
