### 3. Implement `_load_words()`
```gdscript
func _load_words() -> void:
    if not FileAccess.file_exists(WORD_FILE_PATH):
        push_warning("WordValidator: word file missing, using fallback list")
        for w in FALLBACK_WORDS:
            _words[w] = true
        _loaded = true
        return

    var file := FileAccess.open(WORD_FILE_PATH, FileAccess.READ)
    while not file.eof_reached():
        var word := file.get_line().strip_edges()
        if word.length() >= 3:
            _words[word] = true
    file.close()
    _loaded = true
    print("WordValidator: loaded %d words" % _words.size())
```

### 4. Implement `is_valid(word: String) -> bool`
```gdscript
func is_valid(word: String) -> bool:
    return _words.has(word.to_lower())
```

### 5. Implement `validate_submission(tiles: Array)`
```gdscript
func validate_submission(tiles: Array) -> void:
    var word := "".join(tiles.map(func(t): return t.letter))
    if is_valid(word):
        var raw_score := _base_score(word.length())
        emit_signal("word_valid", word, tiles, raw_score)
    else:
        emit_signal("word_invalid", word)
```

### 6. Implement `_base_score(length: int) -> int`
```gdscript
func _base_score(length: int) -> int:
    match length:
        3: return 10
        4: return 25
        5: return 50
        _: return 100   # 6+ letters
```

## Notes
- `O(1)` lookup: `Dictionary.has()` in GDScript is a hash-map probe — far faster than `Array.has()` on 50k entries
- Raw score emitted with `word_valid` is the BASE score only; ScoreManager (T014) applies combo multiplier and letter bonus on top
- WordValidator is stateless after load — it does not track combo or session context
