## Steps

### 1. Replace stub `scripts/autoloads/SaveGame.gd` with full implementation

```gdscript
extends Node

const SAVE_PATH := "user://save.json"

const DEFAULTS: Dictionary = {
    "best_score":      0,
    "selected_theme":  "default",
    "sound_enabled":   true,
    "unlocked_themes": [],
    "daily":           {}   # { "YYYYMMDD": score }
}

var _data: Dictionary = {}
```

### 2. Implement `_ready()`
```gdscript
func _ready() -> void:
    _load()
```

### 3. Implement `_load()`
```gdscript
func _load() -> void:
    if not FileAccess.file_exists(SAVE_PATH):
        _data = DEFAULTS.duplicate(true)
        return
    var file := FileAccess.open(SAVE_PATH, FileAccess.READ)
    var text := file.get_as_text()
    file.close()
    var parsed := JSON.parse_string(text)
    if parsed == null or not parsed is Dictionary:
        push_warning("SaveGame: corrupt save file, resetting to defaults")
        _data = DEFAULTS.duplicate(true)
        return
    # Merge with defaults to handle missing keys from older save versions
    _data = DEFAULTS.duplicate(true)
    for key in parsed:
        _data[key] = parsed[key]
```

### 4. Implement `_save()`
```gdscript
func _save() -> void:
    var file := FileAccess.open(SAVE_PATH, FileAccess.WRITE)
    file.store_string(JSON.stringify(_data, "\t"))
    file.close()
```

### 5. Implement public API

```gdscript
func get_best_score() -> int:
    return _data.get("best_score", 0)

func set_best_score(n: int) -> void:
    if n > _data.get("best_score", 0):
        _data["best_score"] = n
        _save()

func get_theme() -> String:
    return _data.get("selected_theme", "default")

func set_theme(name: String) -> void:
    _data["selected_theme"] = name
    _save()

func get_sound_enabled() -> bool:
    return _data.get("sound_enabled", true)

func set_sound_enabled(value: bool) -> void:
    _data["sound_enabled"] = value
    _save()

func is_theme_unlocked(name: String) -> bool:
    return name == "default" or name in _data.get("unlocked_themes", [])

func unlock_theme(name: String) -> void:
    var unlocked: Array = _data.get("unlocked_themes", [])
    if not name in unlocked:
        unlocked.append(name)
        _data["unlocked_themes"] = unlocked
        _save()

func get_daily_result(date_str: String) -> int:
    return _data.get("daily", {}).get(date_str, -1)

func set_daily_result(date_str: String, score: int) -> void:
    if not _data.has("daily"):
        _data["daily"] = {}
    _data["daily"][date_str] = score
    _save()
```

### 7. Commit `scripts/autoloads/SaveGame.gd`

## Acceptance Criteria
- First run: `user://save.json` is created with defaults on first write (or loaded from file on subsequent runs)
- `set_best_score(500)` followed by `get_best_score()` returns `500`; kill and relaunch app — `get_best_score()` still returns `500`
- `set_theme("ocean")` → `get_theme()` returns `"ocean"` after restart
- `set_daily_result("20260509", 350)` → `get_daily_result("20260509")` returns `350` after restart
- Corrupt the save file manually (write invalid JSON) → game resets to defaults without crashing
- `is_theme_unlocked("ocean")` returns `false` (not in unlocked_themes); after `unlock_theme("ocean")`, returns `true`
- No GDScript errors; committed

## Notes
- `_data` is an in-memory copy — reads are free; writes always flush to disk immediately (acceptable given write frequency: at most one write per word submission + one per game over)
- Do NOT call `_save()` in `_ready()` — only save when data actually changes
- `user://` maps to Android's internal app storage — survives app updates, cleared on uninstall
